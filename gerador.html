function buscarLinkVideosDrive(id) { fetch(API_URL + "?evento=" + encodeURIComponent(id)) .then(res => res.json()) .then(dados => { if (dados.configuracoes) { var link = dados.configuracoes.link_videos || ""; if (link && String(link).trim() !== "") { linkDriveVideos = String(link).trim(); document.getElementById('drive-url-display').value = linkDriveVideos; document.getElementById('btn-open-drive').href = linkDriveVideos; document.getElementById('drive-box').style.display = 'block'; } else { document.getElementById('drive-box').style.display = 'none'; } } }) .catch(err => console.error("Erro ao buscar link de vídeos:", err)); }
